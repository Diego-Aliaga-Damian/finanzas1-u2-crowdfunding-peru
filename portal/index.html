Reproducir todo · PowerShell 
# 1. Descargar el proyecto y preparar el entorno
git clone 
cd finanzas1-u2-crowdfunding-peru
python -m venv .venv ; .venv\Scripts\activate
pip install -r requirements.txt
# 2. Extraer, limpiar y analizar (de las fuentes oficiales a las tablas)
python codigo/01_extraccion_api.py
python codigo/02_scraping_web.py
python codigo/03_limpieza_datos.py
python codigo/04_analisis.py
# 3. Verificar
python -m pytest
python codigo/auditoria.py --en-vivo
